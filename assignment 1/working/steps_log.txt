frames = []
for table, mapping in keep.items():
    df = data_sources[table].copy()
    df["item"] = df["item"].str.replace(r"\s+", " ", regex=True).str.strip()   # fix double spaces

    missing = set(mapping) - set(df["item"])          # labels not found in the data
    if missing:
        print(f"WARNING {table}: not found -> {missing}")

    df = df[df["item"].isin(mapping)].copy()
    df["series"] = df["item"].map(mapping)
    df["source"], df["table"], df["is_forecast"] = "CSO", table.upper(), False
    frames.append(df)
    print(f"{table} : {df.shape[0]} rows <- {len(mapping)} series x {df['year'].nunique()} years "
          f"({df['year'].min()}-{df['year'].max()})")

for i in sorted(data_location.glob("*.xlsx")):
    # skip Excel's temporary lock files (created while a workbook is open)
    if i.name.startswith("~$"):
        continue
    # get the pack year from the file name, e.g. "...June-2026.xlsx" -> 2026
    found = re.search(r"20\d{2}", i.name)
    if not found:
        print(f"skipped (no year in name): {i.name}")
        continue
    pack_year = int(found.group())
    # only extract packs whose tab positions have been defined
    if pack_year not in ifac_tabs_by_year:
        print(f"skipped (no tab positions defined for {pack_year}): {i.name}")
        continue
    # extract each tab into its own dataframe and store it in the dictionary
    for tab_key, (sheet, year_row, first_col, unit, scale, rows) in ifac_tabs_by_year[pack_year].items():
        df = ifac_block(i, pack_year, tab_key, sheet, year_row, first_col, unit, scale, rows)
        key = f"ifac{pack_year}_{tab_key}"
        data_sources[key] = df
        print(f"{key} : {df.shape} <-{i.name} [{sheet}]")
